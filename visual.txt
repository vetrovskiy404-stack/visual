--==================== СОХРАНЕНИЕ ====================
local function ser(v)
	if typeof(v) == "Color3" then
		return { __c = v:ToHex() }
	elseif type(v) == "table" then
		local o = {}
		for k, x in pairs(v) do o[k] = ser(x) end
		return o
	end
	return v
end

local function applyData(dst, src)
	for k, v in pairs(src) do
		local cur = dst[k]
		if type(v) == "table" then
			if v.__c then
				local ok, col = pcall(Color3.fromHex, v.__c)
				if ok and typeof(cur) == "Color3" then dst[k] = col end
			elseif type(cur) == "table" then
				applyData(cur, v)
			end
		elseif cur ~= nil and typeof(cur) == typeof(v) then
			dst[k] = v
		end
	end
end

local FILE = "BBVisuals_v2.json"
local hasFS = (writefile and readfile and isfile) and true or false
local defaults = ser(cfg)

local function saveCfg()
	if not hasFS then return "Файлы недоступны в этом окружении" end
	local ok, err = pcall(function() writefile(FILE, HttpService:JSONEncode(ser(cfg))) end)
	return ok and "Сохранено" or ("Ошибка: " .. tostring(err))
end

local function loadCfg()
	if not hasFS then return "Файлы недоступны в этом окружении" end
	if not isfile(FILE) then return "Сохранения нет" end
	local ok, err = pcall(function() applyData(cfg, HttpService:JSONDecode(readfile(FILE))) end)
	return ok and "Загружено" or ("Ошибка: " .. tostring(err))
end

if hasFS then pcall(function() if isfile(FILE) then loadCfg() end end) end

local mapBounds = { cx = 0, cz = 0, minY = 0, sx = 300, sz = 300, found = false }
local computing = false
local layers = {}

local function updateMapLabels(msg)
	for _, l in ipairs(mapLabels) do l.Text = msg end
end

local function computeBounds()
	if computing then return end
	computing = true
	updateMapLabels("Считаю границы карты...")
	task.spawn(function()
		local root = Workspace:FindFirstChild("Map") or Workspace
		local minX, minY, minZ = math.huge, math.huge, math.huge
		local maxX, maxZ = -math.huge, -math.huge
		local n = 0
		for i, d in ipairs(root:GetDescendants()) do
			if d:IsA("BasePart") and d.Anchored and not d:IsA("Terrain") and d.Transparency < 1
				and d.Size.Magnitude < 1500 and d.Name ~= "ParticleField"
				and not (d.Parent and (d.Parent.Name == "Balls" or d.Parent:FindFirstChildOfClass("Humanoid"))) then
				local cf, s = d.CFrame, d.Size
				local ex = (math.abs(cf.RightVector.X) * s.X + math.abs(cf.UpVector.X) * s.Y + math.abs(cf.LookVector.X) * s.Z) / 2
				local ez = (math.abs(cf.RightVector.Z) * s.X + math.abs(cf.UpVector.Z) * s.Y + math.abs(cf.LookVector.Z) * s.Z) / 2
				local ey = (math.abs(cf.RightVector.Y) * s.X + math.abs(cf.UpVector.Y) * s.Y + math.abs(cf.LookVector.Y) * s.Z) / 2
				local p = cf.Position
				minX = math.min(minX, p.X - ex); maxX = math.max(maxX, p.X + ex)
				minZ = math.min(minZ, p.Z - ez); maxZ = math.max(maxZ, p.Z + ez)
				minY = math.min(minY, p.Y - ey)
				n += 1
			end
			if i % 400 == 0 then task.wait() end
		end
		if n > 0 then
			mapBounds.cx = (minX + maxX) / 2
			mapBounds.cz = (minZ + maxZ) / 2
			mapBounds.minY = minY
			mapBounds.sx = math.clamp(maxX - minX, 60, 1500)
			mapBounds.sz = math.clamp(maxZ - minZ, 60, 1500)
			mapBounds.found = true
			updateMapLabels(string.format("Карта: %d x %d (частей: %d)", mapBounds.sx, mapBounds.sz, n))
		else
			updateMapLabels("Карту найти не удалось, использую область 300 x 300")
		end
		computing = false
		for _, L in ipairs(layers) do applyLayer(L) end
	end)
end

local function texId(c)
	if c.textureIndex == #textures then
		local d = tostring(c.customId):gsub("%D", "")
		if d ~= "" then return "rbxassetid://" .. d end
		return textures[1][2]
	end
	return textures[c.textureIndex][2]
end

applyLayer = function(L)
	local c, pe = L.cfg, L.pe
	pe.Enabled = c.enabled
	pe.Brightness = c.brightness
	pe.LightEmission = c.glow
	pe.LightInfluence = c.lightInfluence
	pe.Lifetime = NumberRange.new(c.lifetime * 0.6, c.lifetime)
	pe.Speed = NumberRange.new(c.speed * 0.4, c.speed)
	pe.Drag = c.drag
	pe.Acceleration = Vector3.new(c.windX, c.windY, c.windZ)
	pe.SpreadAngle = Vector2.new(c.spread, c.spread)
	pe.RotSpeed = c.rotate and NumberRange.new(-c.rotSpeed, c.rotSpeed) or NumberRange.new(0, 0)
	pe.ZOffset = c.zOffset
	pe.TimeScale = c.timeScale
	pe.Orientation = orientEnums[c.orientation]
	pe.Texture = texId(c)

	local s, v = c.size, c.size * c.sizeVar
	local K = NumberSequenceKeypoint.new
	if c.sizeMode == 1 then
		pe.Size = NumberSequence.new({ K(0, s, v), K(1, s, v) })
	elseif c.sizeMode == 2 then
		pe.Size = NumberSequence.new({ K(0, 0, 0), K(1, s, v) })
	elseif c.sizeMode == 3 then
		pe.Size = NumberSequence.new({ K(0, s, v), K(1, 0, 0) })
	else
		pe.Size = NumberSequence.new({ K(0, 0, 0), K(0.5, s, v), K(1, 0, 0) })
	end

	local f = math.clamp(c.fadeIn, 0.02, 0.45)
	pe.Transparency = NumberSequence.new({ K(0, 1), K(f, c.transparency), K(1 - f, c.transparency), K(1, 1) })

	if not c.rainbow then
		pe.Color = c.gradient and ColorSequence.new(c.color, c.color2) or ColorSequence.new(c.color)
	end

	if c.mode == 3 then
		L.part.Size = Vector3.new(mapBounds.sx, c.height, mapBounds.sz)
		local mult = math.clamp((mapBounds.sx * mapBounds.sz) / math.max(c.area * c.area, 1), 1, c.densityCap)
		pe.Rate = math.min(c.rate * mult, 5000)
	else
		L.part.Size = Vector3.new(c.area, c.height, c.area)
		pe.Rate = c.rate
	end
end

local function applyLayers()
	for _, L in ipairs(layers) do applyLayer(L) end
end

--==================== МЯЧ: ТРЕЙЛ + ЭФФЕКТЫ ====================
local trails = {} -- [ball] = {trail,a0,a1,hl,light,sp}
local curBall = nil

local function applyBallTo(t)
	local c, b = cfg.trail, cfg.ball
	local K = NumberSequenceKeypoint.new
	local tr = t.trail
	tr.Enabled = c.enabled
	tr.Lifetime = c.lifetime
	tr.MinLength = c.minLength
	tr.LightEmission = c.glow
	tr.FaceCamera = c.faceCamera
	tr.Texture = trailTexPaths[c.textureIndex]
	tr.TextureMode = Enum.TextureMode.Stretch
	tr.TextureLength = c.textureLength
	tr.WidthScale = NumberSequence.new({ K(0, c.width), K(1, 0) })
	tr.Transparency = NumberSequence.new({ K(0, c.transparency), K(1, 1) })
	if not c.rainbow then
		tr.Color = c.gradient and ColorSequence.new(c.color, c.color2) or ColorSequence.new(c.color)
	end

	t.hl.Enabled = b.hl
	t.hl.FillColor = b.hlFill
	t.hl.OutlineColor = b.hlOutline
	t.hl.FillTransparency = b.hlFillT
	t.hl.OutlineTransparency = b.hlOutlineT

	t.light.Enabled = b.light
	t.light.Color = b.lightColor
	t.light.Brightness = b.lightBrightness
	t.light.Range = b.lightRange

	t.sp.Enabled = b.sparks
	t.sp.Rate = b.sparksRate
	t.sp.Size = NumberSequence.new({ K(0, b.sparksSize), K(1, 0) })
	t.sp.Color = ColorSequence.new(b.sparksColor)
end

local function applyBalls()
	for ball, t in pairs(trails) do
		if t.trail.Parent then applyBallTo(t) else trails[ball] = nil end
	end
end

local function attachBall(ball)
	if not ball:IsA("BasePart") or trails[ball] then return end
	local a0 = Instance.new("Attachment")
	a0.Name = "BBA0"
	a0.Position = Vector3.new(0, 0.5, 0)
	a0.Parent = ball
	local a1 = Instance.new("Attachment")
	a1.Name = "BBA1"
	a1.Position = Vector3.new(0, -0.5, 0)
	a1.Parent = ball

	local trail = Instance.new("Trail")
	trail.Name = "BBTrail"
	trail.Attachment0 = a0
	trail.Attachment1 = a1
	trail.Parent = ball

	local hl = Instance.new("Highlight")
	hl.Name = "BBHighlight"
	hl.Adornee = ball
	hl.Parent = ball

	local light = Instance.new("PointLight")
	light.Name = "BBLight"
	light.Parent = ball

	local sp = Instance.new("ParticleEmitter")
	sp.Name = "BBSparks"
	sp.Texture = textures[1][2]
	sp.LightEmission = 1
	sp.Brightness = 3
	sp.Lifetime = NumberRange.new(0.4, 0.8)
	sp.Speed = NumberRange.new(2, 6)
	sp.SpreadAngle = Vector2.new(180, 180)
	sp.Parent = ball

	trails[ball] = { trail = trail, a0 = a0, a1 = a1, hl = hl, light = light, sp = sp }
	applyBallTo(trails[ball])
end

local function findBall()
	local folder = Workspace:FindFirstChild("Balls")
	if not folder then return nil end
	local best
	for _, b in ipairs(folder:GetChildren()) do
		if b:IsA("BasePart") then
			if b:GetAttribute("realBall") then return b end
			best = best or b
		end
	end
	return best
end

task.spawn(function()
	while running do
		local folder = Workspace:FindFirstChild("Balls")
		if folder then
			for _, b in ipairs(folder:GetChildren()) do attachBall(b) end
		end
		curBall = findBall()
		task.wait(0.25)
	end
end)

--==================== ЦИКЛ ОБНОВЛЕНИЯ ====================
local smoothFov
bind(RunService.RenderStepped, function(dt)
	local cam = Workspace.CurrentCamera
	if not cam then return end

	-- FOV (+ динамический от скорости мяча)
	if cfg.cam.fovEnabled then
		smoothFov = smoothFov or cam.FieldOfView
		local target = cfg.cam.fov
		if cfg.cam.dyn and curBall then
			target += math.clamp(curBall.AssemblyLinearVelocity.Magnitude / 150, 0, 1) * cfg.cam.dynAmount
		end
		smoothFov += (target - smoothFov) * math.clamp(dt * 6, 0, 1)
		cam.FieldOfView = smoothFov
	else
		smoothFov = nil
	end

	-- положение полей частиц
	local camPos = cam.CFrame.Position
	local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
	local hue = (os.clock() * 0.2) % 1
	for _, L in ipairs(layers) do
		local c = L.cfg
		if c.enabled then
			if c.mode == 3 then
				L.part.CFrame = CFrame.new(mapBounds.cx, mapBounds.minY + c.height / 2 + c.offsetY, mapBounds.cz)
			else
				local pos = camPos
				if c.mode == 2 and hrp then pos = hrp.Position
				elseif c.mode == 4 and curBall then pos = curBall.Position end
				L.part.CFrame = CFrame.new(pos + Vector3.new(0, c.offsetY, 0))
			end
			if c.rainbow then
				L.pe.Color = c.gradient
					and ColorSequence.new(Color3.fromHSV(hue, 0.7, 1), Color3.fromHSV((hue + 0.3) % 1, 0.7, 1))
					or ColorSequence.new(Color3.fromHSV(hue, 0.7, 1))
			end
		end
	end

	-- радужный трейл
	if cfg.trail.rainbow then
		local seq = ColorSequence.new(Color3.fromHSV(hue, 1, 1), Color3.fromHSV((hue + 0.25) % 1, 1, 1))
		for _, t in pairs(trails) do
			if t.trail.Parent then t.trail.Color = seq end
		end
	end

	-- игра может менять освещение сама, поэтому держим свои значения
	local w = cfg.world
	if w.clock then Lighting.ClockTime = w.clockTime end
	if w.bright then Lighting.Brightness = w.brightness end
	if w.fog then
		Lighting.FogStart, Lighting.FogEnd, Lighting.FogColor = w.fogStart, w.fogEnd, w.fogColor
	end
end)

local function randIn(a, b) return a + (b - a) * math.random() end

local function randUnit()
	local v = Vector3.new(math.random() * 2 - 1, math.random() * 2 - 1, math.random() * 2 - 1)
	if v.Magnitude < 0.05 then return Vector3.yAxis end
	return v.Unit
end

local function pickColor(c)
	if c.rainbow then return Color3.fromHSV(math.random(), 0.7, 1) end
	return c.color:Lerp(c.color2, math.random())
end

local function sizeAt(mode, t)
	if mode == 1 then return 1 elseif mode == 2 then return t elseif mode == 3 then return 1 - t end
	return math.sin(t * math.pi)
end

local function mapMult(c)
	return math.clamp((mapBounds.sx * mapBounds.sz) / math.max(c.area * c.area, 1), 1, c.densityCap)
end

local function matFor(c)
	return c.neon and Enum.Material.Neon or Enum.Material.SmoothPlastic
end

local function buildPool(S)
	for _, p in ipairs(S.all) do p:Destroy() end
	S.all, S.free, S.active = {}, {}, {}
	local c = S.cfg
	local shape = (c.kind == 2) and Enum.PartType.Ball or Enum.PartType.Block
	for i = 1, c.maxCount do
		local p = newPart(shape)
		S.all[i] = p
		S.free[i] = { part = p }
	end
end

local function ensureSplash(S)
	if #S.rings > 0 then return end
	for i = 1, 30 do
		local p = newPart(Enum.PartType.Cylinder)
		p.Size = Vector3.new(0.03, 1, 1)
		S.rings[i] = { part = p, active = false, age = 0, life = 0.35, maxD = 1 }
	end
	for i = 1, 100 do
		S.sprayFree[i] = { part = newPart(Enum.PartType.Ball) }
	end
end

-- точка появления частицы (и скорость источника для мяча/игрока)
local function spawnPoint(S, camPos, hrp)
	local c = S.cfg
	if S.src == "ball" then
		if not curBall then return nil end
		return curBall.Position + Vector3.new(0, c.offsetY, 0) + randUnit() * 0.5, curBall.AssemblyLinearVelocity
	elseif S.src == "player" then
		if not hrp then return nil end
		return hrp.Position + Vector3.new(0, c.offsetY, 0) + randUnit() * 0.5, hrp.AssemblyLinearVelocity
	end

	local base, x, z
	if c.mode == 3 then
		base = mapBounds.minY
		x = mapBounds.cx + randIn(-mapBounds.sx / 2, mapBounds.sx / 2)
		z = mapBounds.cz + randIn(-mapBounds.sz / 2, mapBounds.sz / 2)
	else
		local pos = camPos
		if c.mode == 2 and hrp then pos = hrp.Position
		elseif c.mode == 4 and curBall then pos = curBall.Position end
		local a, r = math.random() * math.pi * 2, c.area * math.sqrt(math.random())
		base = pos.Y
		x = pos.X + math.cos(a) * r
		z = pos.Z + math.sin(a) * r
	end
	local y
	if c.kind == 1 then
		y = base + c.height + c.offsetY
	elseif c.mode == 3 then
		y = base + c.offsetY + randIn(0, c.height)
	else
		y = base + c.offsetY + randIn(-c.height / 2, c.height / 2)
	end
	return Vector3.new(x, y, z), Vector3.zero
end

-- капля: один луч до земли при появлении, потом честное падение и всплеск в точке попадания
local function spawnDrop(S, origin)
	local c = S.cfg
	local e = table.remove(S.free)
	if not e then return end
	local speed = math.max(c.fallSpeed * randIn(0.85, 1.15), 1)
	local vel = Vector3.new(c.windX + randIn(-1, 1) * c.spread, -speed, c.windZ + randIn(-1, 1) * c.spread)
	local dir = vel.Unit
	local maxDown = (S.src == "box") and (c.height + 120) or 300
	local rayLen = math.min(maxDown / math.max(math.abs(dir.Y), 0.25), 800)
	local hit = Workspace:Raycast(origin, dir * rayLen, rayParams)

	e.kind, e.pos, e.vel, e.dir, e.age = 1, origin, vel, dir, 0
	if hit then
		e.impactT = (hit.Position - origin).Magnitude / vel.Magnitude
		e.hitPos, e.hitN = hit.Position, hit.Normal
	else
		e.impactT = nil
		e.life = rayLen / vel.Magnitude
	end

	local w = math.max(c.size * (1 + randIn(-c.sizeVar, c.sizeVar)), 0.03)
	e.w = w
	e.len = math.max(vel.Magnitude * 0.045 * c.length, w)
	e.color = pickColor(c)
	local p = e.part
	p.Size = Vector3.new(w, w, e.len)
	p.Color = e.color
	p.Material = matFor(c)
	p.Transparency = c.transparency
	S.active[#S.active + 1] = e
end

local function spawnCircle(S, origin, srcVel)
	local c = S.cfg
	local e = table.remove(S.free)
	if not e then return end
	local vel = randUnit() * (c.speed * randIn(0.3, 1)) + Vector3.new(c.windX, c.windY, c.windZ)
	if S.src ~= "box" then
		vel = vel + srcVel * c.inherit + randUnit() * c.spread
	end
	e.kind, e.pos, e.vel, e.age = 2, origin, vel, 0
	e.life = math.max(c.lifetime * randIn(0.6, 1), 0.2)
	e.size = math.max(c.size * (1 + randIn(-c.sizeVar, c.sizeVar)), 0.03)
	e.mode = c.sizeMode
	local p = e.part
	local s0 = math.max(e.size * sizeAt(e.mode, 0), 0.02)
	p.Size = Vector3.new(s0, s0, s0)
	p.Color = pickColor(c)
	p.Material = matFor(c)
	p.Transparency = c.transparency
	S.active[#S.active + 1] = e
end

local function surfaceCF(pos, n)
	local t = n:Cross(Vector3.xAxis)
	if t.Magnitude < 0.1 then t = n:Cross(Vector3.zAxis) end
	t = t.Unit
	return CFrame.fromMatrix(pos, n, t), t, n:Cross(t)
end

-- всплеск: расширяющееся пятно на поверхности + брызги, летящие по дуге
local function splash(S, e)
	local c = S.cfg
	if not (c.splashRing or c.spray) then return end
	local n = e.hitN
	local pos = e.hitPos + n * 0.03
	local cf, t, b = surfaceCF(pos, n)
	ensureSplash(S)

	if c.splashRing and n.Y > 0.5 then
		S.ringIdx = S.ringIdx % #S.rings + 1
		local r = S.rings[S.ringIdx]
		r.active, r.age, r.life = true, 0, 0.35
		r.maxD = math.max(c.splashSize * randIn(0.7, 1.2), 0.2)
		r.cf = cf
		r.part.Color = e.color
		r.part.Material = matFor(c)
	end

	if c.spray then
		local sz = math.max(e.w * 1.5, 0.06)
		for _ = 1, c.sprayCount do
			local sp = table.remove(S.sprayFree)
			if not sp then break end
			local a = math.random() * math.pi * 2
			sp.pos = pos
			sp.vel = n * randIn(3, 7) + (t * math.cos(a) + b * math.sin(a)) * randIn(0.8, 3.2)
			sp.age, sp.life = 0, randIn(0.2, 0.38)
			sp.part.Size = Vector3.new(sz, sz, sz)
			sp.part.Color = e.color
			sp.part.Material = matFor(c)
			sp.part.Transparency = c.transparency
			S.sprayActive[#S.sprayActive + 1] = sp
		end
	end
end

local plist, clist = {}, {}

local function stepSys(S, dt, camPos, hrp)
	local c = S.cfg

	local act = S.active

	if not c.enabled then
		-- выключено: убираем все детали, пул освобождаем
		if #S.all > 0 then
			for _, p in ipairs(S.all) do p:Destroy() end
			S.all, S.free, S.active = {}, {}, {}
			act = S.active
		end
		S.sig, S.poolDirty = nil, false
	else
		-- пересборка пула при смене вида или максимума (с задержкой, чтобы не мучить ползунком)
		local sig = c.kind * 100000 + c.maxCount
		if sig ~= S.sig then
			S.sig = sig
			S.poolDirty, S.poolDirtyAt = true, os.clock()
		end
		if S.poolDirty and os.clock() - S.poolDirtyAt > 0.4 then
			buildPool(S)
			act = S.active
			S.builtKind = c.kind
			S.poolDirty = false
		end
	end

	if c.enabled and not S.poolDirty and S.builtKind == c.kind then
		local rate = c.rate
		if S.src == "box" and c.mode == 3 then rate = rate * mapMult(c) end
		S.acc = math.min(S.acc + rate * dt, 40)
		while S.acc >= 1 do
			S.acc -= 1
			local pt, sv = spawnPoint(S, camPos, hrp)
			if not pt then
				S.acc = 0
				break
			end
			if c.kind == 1 then spawnDrop(S, pt) else spawnCircle(S, pt, sv) end
		end
	end

	-- обновление живых частиц
	for i = #act, 1, -1 do
		local e = act[i]
		e.age += dt
		local dead = false
		if e.kind == 1 then
			e.pos = e.pos + e.vel * dt
			if e.impactT and e.age >= e.impactT then
				splash(S, e)
				dead = true
			elseif not e.impactT and e.age >= e.life then
				dead = true
			else
				local center = e.pos - e.dir * (e.len * 0.5)
				plist[#plist + 1] = e.part
				clist[#clist + 1] = CFrame.lookAt(center, center + e.dir, math.abs(e.dir.Y) > 0.99 and Vector3.xAxis or Vector3.yAxis)
			end
		else
			local t = e.age / e.life
			if t >= 1 then
				dead = true
			else
				e.pos = e.pos + e.vel * dt
				if e.mode ~= 1 then
					local s = math.max(e.size * sizeAt(e.mode, t), 0.02)
					e.part.Size = Vector3.new(s, s, s)
				end
				plist[#plist + 1] = e.part
				clist[#clist + 1] = CFrame.new(e.pos)
			end
		end
		if dead then
			plist[#plist + 1] = e.part
			clist[#clist + 1] = HIDE
			act[i] = act[#act]
			act[#act] = nil
			S.free[#S.free + 1] = e
		end
	end

	-- брызги
	local sa = S.sprayActive
	for i = #sa, 1, -1 do
		local sp = sa[i]
		sp.age += dt
		if sp.age >= sp.life then
			plist[#plist + 1] = sp.part
			clist[#clist + 1] = HIDE
			sa[i] = sa[#sa]
			sa[#sa] = nil
			S.sprayFree[#S.sprayFree + 1] = sp
		else
			sp.vel = sp.vel + Vector3.new(0, -60 * dt, 0)
			sp.pos = sp.pos + sp.vel * dt
			plist[#plist + 1] = sp.part
			clist[#clist + 1] = CFrame.new(sp.pos)
		end
	end

	-- пятна на поверхности
	for _, r in ipairs(S.rings) do
		if r.active then
			r.age += dt
			local t = r.age / r.life
			if t >= 1 then
				r.active = false
				plist[#plist + 1] = r.part
				clist[#clist + 1] = HIDE
			else
				local d = r.maxD * (0.15 + 0.85 * t)
				r.part.Size = Vector3.new(0.03, d, d)
				r.part.Transparency = c.transparency + (1 - c.transparency) * t
				plist[#plist + 1] = r.part
				clist[#clist + 1] = r.cf
			end
		end
	end
end

local ptObj = {}

local function applyPlayerTrail()
	local c = cfg.ptrail
	if not (ptObj.trail and ptObj.trail.Parent) then return end
	local K = NumberSequenceKeypoint.new
	ptObj.a0.Position = Vector3.new(0, c.offsetY + c.height / 2, 0)
	ptObj.a1.Position = Vector3.new(0, c.offsetY - c.height / 2, 0)
	local tr = ptObj.trail
	tr.Enabled = c.enabled
	tr.Lifetime = c.lifetime
	tr.MinLength = c.minLength
	tr.LightEmission = c.glow
	tr.FaceCamera = c.faceCamera
	tr.Texture = trailTexPaths[c.textureIndex]
	tr.TextureMode = Enum.TextureMode.Stretch
	tr.TextureLength = c.textureLength
	tr.WidthScale = NumberSequence.new({ K(0, 1), K(1, 0) })
	tr.Transparency = NumberSequence.new({ K(0, c.transparency), K(1, 1) })
	if not c.rainbow then
		tr.Color = c.gradient and ColorSequence.new(c.color, c.color2) or ColorSequence.new(c.color)
	end
end

local function ensurePlayerTrail()
	local ch = lp.Character
	local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	if ptObj.hrp == hrp and ptObj.trail and ptObj.trail.Parent then return end
	for _, k in ipairs({ "trail", "a0", "a1" }) do
		local o = ptObj[k]
		if o then pcall(function() o:Destroy() end) end
	end
	local a0 = Instance.new("Attachment")
	a0.Name = "BBPA0"
	a0.Parent = hrp
	local a1 = Instance.new("Attachment")
	a1.Name = "BBPA1"
	a1.Parent = hrp
	local tr = Instance.new("Trail")
	tr.Name = "BBPlayerTrail"
	tr.Attachment0 = a0
	tr.Attachment1 = a1
	tr.Parent = hrp
	ptObj.hrp, ptObj.a0, ptObj.a1, ptObj.trail = hrp, a0, a1, tr
	applyPlayerTrail()
end

local function Slider(parent, name, min, max, dec, tbl, key, onChange)
	local f = row(parent, 48)
	text(f, name, 13, UDim2.new(0, 12, 0, 4), UDim2.new(0.6, 0, 0, 20))
	local val = text(f, "", 13, UDim2.new(0.6, 0, 0, 4), UDim2.new(0.4, -12, 0, 20), Enum.TextXAlignment.Right)
	val.TextColor3 = ACCENT

	local hit = Instance.new("Frame")
	hit.Size = UDim2.new(1, -24, 0, 22)
	hit.Position = UDim2.new(0, 12, 0, 24)
	hit.BackgroundTransparency = 1
	hit.Parent = f
	local track = Instance.new("Frame")
	track.Name = "Track"
	track.Size = UDim2.new(1, 0, 0, 6)
	track.Position = UDim2.new(0, 0, 0.5, -3)
	track.BackgroundColor3 = Color3.new(1, 1, 1)
	track.BackgroundTransparency = 0.8
	track.BorderSizePixel = 0
	track.Parent = hit
	corner(track, 3)
	local fill = Instance.new("Frame")
	fill.BackgroundColor3 = ACCENT
	fill.BorderSizePixel = 0
	fill.Parent = track
	corner(fill, 3)
	local knob = Instance.new("Frame")
	knob.Size = UDim2.new(0, 12, 0, 12)
	knob.AnchorPoint = Vector2.new(0.5, 0.5)
	knob.BackgroundColor3 = Color3.new(1, 1, 1)
	knob.BorderSizePixel = 0
	knob.Parent = track
	corner(knob, 6)

	local function show(v)
		local a = (v - min) / (max - min)
		fill.Size = UDim2.new(a, 0, 1, 0)
		knob.Position = UDim2.new(a, 0, 0.5, 0)
		val.Text = tostring(v)
	end
	local function set(v)
		v = math.clamp(v, min, max)
		local m = 10 ^ dec
		v = math.floor(v * m + 0.5) / m
		tbl[key] = v
		show(v)
		if onChange then onChange() end
	end
	local function fromX(x)
		local a = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
		set(min + (max - min) * a)
	end

	local dragging = false
	hit.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			fromX(i.Position.X)
		end
	end)
	bind(UIS.InputChanged, function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			fromX(i.Position.X)
		end
	end)
	bind(UIS.InputEnded, function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	show(math.clamp(tbl[key], min, max))
	refreshers[#refreshers + 1] = function() show(math.clamp(tbl[key], min, max)) end
end

		h, s, v = tbl[key]:ToHSV()
		render(false)
	end

end

local function applyHud()
	local h, U = cfg.hud, cfg.ui
	hudFrame.Visible = h.enabled and (h.showFps or h.showPing)
	hudFrame.BackgroundColor3 = U.bg
	hudFrame.BackgroundTransparency = h.bgT
	hudScale.Scale = h.scale
	if h.pos == 1 then
		hudFrame.AnchorPoint = Vector2.new(0.5, 0)
		hudFrame.Position = UDim2.new(0.5, 0, 0, 8)
	elseif h.pos == 2 then
		hudFrame.AnchorPoint = Vector2.new(0, 0)
		hudFrame.Position = UDim2.new(0, 64, 0, 8)
	else
		hudFrame.AnchorPoint = Vector2.new(1, 0)
		hudFrame.Position = UDim2.new(1, -12, 0, 8)
	end
	hudLabel.Font = FONT
	hudLabel.TextColor3 = TEXT
end

local function getPing()
	local ok, v = pcall(function() return Stats.Network.ServerStatsItem["Data Ping"]:GetValue() end)
	if ok and type(v) == "number" then return math.floor(v + 0.5) end
	local ok2, v2 = pcall(function() return lp:GetNetworkPing() end)
	if ok2 and type(v2) == "number" then return math.floor(v2 * 1000 + 0.5) end
	return nil
end

local GOOD, MID, BAD = Color3.fromRGB(110, 255, 150), Color3.fromRGB(255, 215, 90), Color3.fromRGB(255, 95, 95)
local function tag(col, s) return string.format('<font color="#%s">%s</font>', col:ToHex(), s) end

local function updateHud(fps)
	local h = cfg.hud
	local parts = {}
	if h.showFps then
		local s = tostring(fps)
		if h.colorCode then s = tag(fps >= 55 and GOOD or (fps >= 30 and MID or BAD), s) end
		parts[#parts + 1] = "FPS " .. s
	end
	if h.showPing then
		local ping = getPing()
		local s = ping and (tostring(ping) .. " ms") or "-- ms"
		if h.colorCode and ping then s = tag(ping <= 80 and GOOD or (ping <= 150 and MID or BAD), s) end
		parts[#parts + 1] = s
	end
	hudLabel.Text = table.concat(parts, "  •  ")
end

local hudAcc, hudFrames = 0, 0
bind(RunService.RenderStepped, function(dt)
	hudAcc += dt
	hudFrames += 1
	if hudAcc >= 0.5 then
		updateHud(math.floor(hudFrames / hudAcc + 0.5))
		hudAcc, hudFrames = 0, 0
	end
end)

applyTheme = function()
	local U = cfg.ui
	local oldAccent, oldText = ACCENT, TEXT
	ACCENT, TEXT, ROW = U.accent, U.text, U.rowColor
	FONT = fontEnums[U.fontIndex] or fontEnums[1]

	main.Size = UDim2.fromOffset(U.width, U.height)
	main.Position = UDim2.new(0.5, U.posX, 0.5, U.posY)
	mainScale.Scale = U.scale
	mainCorner.CornerRadius = UDim.new(0, U.radius)
	mainStroke.Color = U.strokeColor
	mainStroke.Transparency = U.strokeT
	mainStroke.Thickness = U.strokeW
	main.BackgroundTransparency = U.bgT
	if U.gradient then
		main.BackgroundColor3 = Color3.new(1, 1, 1)
		mainGrad.Color = ColorSequence.new(U.bg, U.bg2)
		mainGrad.Rotation = U.gradRot
		mainGrad.Enabled = true
	else
		main.BackgroundColor3 = U.bg
		mainGrad.Enabled = false
	end

	fab.Visible = U.fab
	fab.BackgroundColor3 = U.bg
	fab.BackgroundTransparency = U.bgT
	fab.TextColor3 = TEXT
	closeBtn.TextColor3 = TEXT
	closeBtn.BackgroundColor3 = ROW
	for _, b in pairs(tabButtons) do b.TextColor3 = TEXT end

	for _, d in ipairs(gui:GetDescendants()) do
		local inPicker = d.Name == "Panel" or d.Name == "Swatch" or d:FindFirstAncestor("Panel") ~= nil
		if d.Name == "Row" then
			d.BackgroundColor3 = ROW
			d.BackgroundTransparency = U.rowT
			local cr = d:FindFirstChildOfClass("UICorner")
			if cr then cr.CornerRadius = UDim.new(0, U.rowRadius) end
		elseif d.Name == "Track" then
			d.BackgroundColor3 = ROW
		elseif d.Name == "Field" then
			d.BackgroundColor3 = ROW
			d.TextColor3 = TEXT
		end
		if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
			if table.find(fontEnums, d.Font) then d.Font = FONT end
			if d.Name ~= "Field" and d.TextColor3 == oldText then d.TextColor3 = TEXT end
		end
		if oldAccent ~= ACCENT and not inPicker then
			if d:IsA("GuiObject") and d.BackgroundColor3 == oldAccent then d.BackgroundColor3 = ACCENT end
			if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.TextColor3 == oldAccent then d.TextColor3 = ACCENT end
			if d:IsA("ScrollingFrame") and d.ScrollBarImageColor3 == oldAccent then d.ScrollBarImageColor3 = ACCENT end
		end
	end

	applyHud()
	selectTab(currentTab or "Частицы 1")
	for _, r in ipairs(refreshers) do pcall(r) end
	applyBlur()
end

		if typeof(v) == "Color3" then return v:ToHex() end
		if type(v) == "table" then
			local keys = {}
			for k in pairs(v) do keys[#keys + 1] = k end
			table.sort(keys, function(a, b) return tostring(a) < tostring(b) end)
			local out = {}
			for _, k in ipairs(keys) do out[#out + 1] = tostring(k) .. "=" .. sig(v[k]) end
			return "{" .. table.concat(out, ",") .. "}"
		end
		return tostring(v)
	end
	local lastSig
	if hasFS then
		local okf, exists = pcall(isfile, FILE)
		if okf and exists then lastSig = sig(cfg) end
	end
	task.spawn(function()
		while running do
			task.wait(2)
			if hasFS and cfg.ui.autosave then
				local cur = sig(cfg)
				if cur ~= lastSig then
					local msg = saveCfg()
					if msg == "Сохранено" then
						lastSig = cur
						st.Text = "Автосохранено в " .. os.date("%H:%M:%S")
					else
						st.Text = msg
					end
				end
			end
		end
	end)

	Section(p, "Конфиг")
	Button(p, "Сохранить настройки", "Сохранить", function() st.Text = saveCfg() end)
	Button(p, "Загрузить настройки", "Загрузить", function()
		st.Text = loadCfg()
		refreshAll()
	end)
	Button(p, "Сбросить всё", "Сбросить", function()
		applyData(cfg, defaults)
		refreshAll()
		st.Text = "Настройки сброшены"
	end)
	Section(p, "Опасная зона")
	Button(p, "Убрать всё и закрыть скрипт", "Выгрузить", function()
		if cfg.ui.autosave then saveCfg() end
		running = false
		for _, cn in ipairs(conns) do pcall(function() cn:Disconnect() end) end
		for _, L in ipairs(layers) do pcall(function() L.part:Destroy() end) end
		for _, t in pairs(trails) do
			for _, o in pairs({ t.trail, t.a0, t.a1, t.hl, t.light, t.sp }) do pcall(function() o:Destroy() end) end
		end
		for _, o in ipairs({ bloom, cc, sun, menuBlur }) do pcall(function() o:Destroy() end) end
		Lighting.Brightness, Lighting.ClockTime = orig.Brightness, orig.ClockTime
		Lighting.FogStart, Lighting.FogEnd, Lighting.FogColor = orig.FogStart, orig.FogEnd, orig.FogColor
		lp.CameraMaxZoomDistance = orig.Zoom
		pcall(function() fxFolder:Destroy() end)
		for _, k in ipairs({ "trail", "a0", "a1" }) do
			local o = ptObj[k]
			if o then pcall(function() o:Destroy() end) end
		end
		gui:Destroy()
	end)
end
